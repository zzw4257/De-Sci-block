合约实现细节

UserProfile 核心函数

function createProfile(string _fullName, uint256 _age, string _email, string _bio)
    external
    onlyVerified
    nonReentrant
    validInput(_fullName, _email)
{
    require(!hasProfile[msg.sender], "Profile exists");
    require(bytes(_fullName).length >= 2, "Name too short");
    require(_age >= 18 && _age <= 120, "Invalid age");

    profiles[msg.sender] = Profile({
        fullName: _fullName,
        age: _age,
        email: _email,
        bio: _bio,
        isVerified: false,
        createdAt: block.timestamp
    });

    emit ProfileCreated(msg.sender, _fullName);
}

Dataset 访问控制

function requestAccess(uint256 _datasetId, string _purpose)
    external
    payable
    nonReentrant
    datasetExists(_datasetId)
{
    Dataset storage dataset = datasets[_datasetId];
    require(dataset.isPublic == false, "Dataset is public");
    require(msg.value >= dataset.accessPrice, "Insufficient payment");

    accessRequests[_datasetId][msg.sender] = AccessRequest({
        purpose: _purpose,
        requestedAt: block.timestamp,
        approved: false
    });

    emit AccessRequested(_datasetId, msg.sender, _purpose);
}
